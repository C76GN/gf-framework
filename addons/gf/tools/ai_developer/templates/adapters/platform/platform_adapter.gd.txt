class_name ProjectPlatformAdapter
extends GFPlatformAdapter


# --- 常量 ---

const CONTRACT_ID: StringName = &"project.platform.sample"
const METHOD_ID: StringName = &"execute"

# --- 私有变量 ---

## 保留项目注入的 SDK 对象；初始化前验证实例有效性，shutdown 清空引用，真实接入须先停止回调入口。
## [br]
## @api private
var _provider: Object = null

## 将 provider 调用 ID 关联到请求句柄，SDK 调用前登记以覆盖同步回调，终态回调取走关联。
## [br]
## @api private
var _provider_calls: Dictionary = {}


# --- 公共方法 ---

func configure_provider(provider: Object, context: GFPlatformRuntimeContext) -> bool:
	_provider = provider
	return configure(
		&"project.sample_adapter",
		&"sample_platform",
		PackedStringArray([CONTRACT_ID]),
		[_make_contract_descriptor()],
		context
	)


func on_provider_succeeded(provider_call_id: String, value: Dictionary) -> void:
	# Native worker callbacks must copy bounded plain data into a project-owned
	# main-thread pump before invoking this method.
	var handle: GFPlatformRequestHandle = _take_handle(provider_call_id)
	if handle != null:
		var _completed: bool = _succeed_request(handle, value)


func on_provider_failed(provider_call_id: String, error_code: StringName) -> void:
	# Translate provider errors to stable, redacted contract status on the main
	# thread; do not expose raw native error text or sensitive identifiers.
	var handle: GFPlatformRequestHandle = _take_handle(provider_call_id)
	if handle != null:
		var _completed: bool = _fail_request(
			handle,
			error_code,
			"Platform provider request failed."
		)


func on_provider_cancelled(provider_call_id: String) -> void:
	# Provider cancellation acknowledgement follows the same main-thread pump
	# boundary as success and failure callbacks.
	var handle: GFPlatformRequestHandle = _take_handle(provider_call_id)
	if handle != null:
		var _released: bool = _release_request(handle)


# --- 可重写钩子 / 虚方法 ---

func _initialize(_options: Dictionary) -> void:
	if not is_instance_valid(_provider):
		var _failed: bool = _fail_initialization("Platform provider is unavailable.")
		return
	# Start the provider here. For asynchronous SDKs, complete from its callback.
	var _completed: bool = _complete_initialization(get_context())


func _dispatch(
	request: GFPlatformBridgeRequest,
	handle: GFPlatformRequestHandle
) -> bool:
	if request.contract_id != CONTRACT_ID or request.method_id != METHOD_ID:
		return false
	# Store a provider call ID -> handle association before invoking the SDK so a
	# synchronous callback cannot escape tracking.
	var provider_call_id: String = String(request.request_id)
	_provider_calls[provider_call_id] = handle
	# Replace this line with one narrow provider call.
	return true


func _cancel_request(handle: GFPlatformRequestHandle, _reason: StringName) -> void:
	if handle == null:
		return
	# Request provider cancellation, but keep the lease until the provider confirms.


func _shutdown() -> void:
	# Before clearing state: stop callback ingress, request cancellation once,
	# detach callbacks, and join Adapter-owned workers within a bounded timeout.
	_provider_calls.clear()
	_provider = null


# --- 私有/辅助方法 ---

## 先移除调用 ID 关联再返回类型正确的句柄，重复或迟到终态回调因此得到 null。
## [br]
## @api private
func _take_handle(provider_call_id: String) -> GFPlatformRequestHandle:
	var value: Variant = GFVariantData.get_option_value(
		_provider_calls,
		provider_call_id
	)
	var _erased: bool = _provider_calls.erase(provider_call_id)
	if value is GFPlatformRequestHandle:
		var handle: GFPlatformRequestHandle = value
		return handle
	return null


## 构造示例静态契约，使用闭合空请求和结果 schema、单并发及明确字节上限；真实适配须替换为实际协议。
## [br]
## @api private
static func _make_contract_descriptor() -> GFPlatformContractDescriptor:
	var request_schema: GFDictionarySchema = GFDictionarySchema.new().configure(
		&"project_platform_sample_request",
		[],
		{"allow_extra_fields": false}
	)
	var result_schema: GFDictionarySchema = GFDictionarySchema.new().configure(
		&"project_platform_sample_result",
		[],
		{"allow_extra_fields": false}
	)
	var method: GFPlatformContractMethodDescriptor = (
		GFPlatformContractMethodDescriptor.new().configure(
			METHOD_ID,
			{
				"request_schema": request_schema,
				"result_schema": result_schema,
				"max_request_bytes": 4096,
				"max_result_bytes": 4096,
				"max_concurrent_requests": 1,
				"supports_cancellation": true,
				"sensitive_fields": PackedStringArray(["token"]),
			}
		)
	)
	return GFPlatformContractDescriptor.new().configure(
		CONTRACT_ID,
		"1.0.0",
		[method]
	)
