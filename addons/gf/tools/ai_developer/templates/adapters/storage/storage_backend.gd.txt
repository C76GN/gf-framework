## 项目所有的 GFStorageBackend 适配骨架，将存储操作委托给单一 ProjectStorageProvider。
## 边界先校验便携逻辑键、闭合响应和有界纯值图，统一错误并限制原子写选项；实际持久化由 Provider 负责。
## [br]
## @api framework_internal
class_name ProjectStorageBackend
extends GFStorageBackend

# --- 可重写钩子 / 虚方法 ---

## 完整校验配置和初始化前后协议/能力，全部成功后才接管 Provider；已调用初始化的失败候选会被关闭。
## 候选提交前没有初始化重入锁；Provider 不得在初始化调用中回调同一后端的初始化或关闭入口。
## [br]
## @api protected
## [br]
## @param config: 项目存储适配器配置。
## [br]
## @schema config: Dictionary，必需 provider:ProjectStorageProvider；可选 provider_options:Dictionary、key_prefix:String，以及 max_payload_bytes、max_list_items、max_list_bytes、max_graph_nodes、max_graph_depth、max_string_bytes、max_container_entries 正整数预算；拒绝其他字段。
## [br]
## @return 初始化提交后为 OK，重复初始化、非法配置或 Provider 协议错误返回对应 Error。
func _initialize(config: Dictionary) -> Error:
	if _initialized or _provider != null:
		return ERR_ALREADY_IN_USE
	if not _has_exact_or_optional_keys(config, _CONFIG_KEYS, ["provider"]):
		return ERR_INVALID_PARAMETER

	var provider_value: Variant = config.get("provider")
	if not provider_value is ProjectStorageProvider:
		return ERR_INVALID_PARAMETER
	var candidate_provider: ProjectStorageProvider = provider_value

	var prefix_value: Variant = config.get("key_prefix", "")
	if not prefix_value is String:
		return ERR_INVALID_PARAMETER
	var requested_prefix: String = prefix_value
	var normalized_prefix: String = _normalize_prefix(requested_prefix)
	if not requested_prefix.is_empty() and normalized_prefix.is_empty():
		return ERR_INVALID_PARAMETER

	var payload_budget: int = _read_bounded_integer(
		config,
		"max_payload_bytes",
		DEFAULT_MAX_PAYLOAD_BYTES,
		ABSOLUTE_MAX_PAYLOAD_BYTES
	)
	var list_item_budget: int = _read_bounded_integer(
		config,
		"max_list_items",
		DEFAULT_MAX_LIST_ITEMS,
		ABSOLUTE_MAX_LIST_ITEMS
	)
	var list_byte_budget: int = _read_bounded_integer(
		config,
		"max_list_bytes",
		DEFAULT_MAX_LIST_BYTES,
		ABSOLUTE_MAX_LIST_BYTES
	)
	var graph_node_budget: int = _read_bounded_integer(
		config,
		"max_graph_nodes",
		DEFAULT_MAX_GRAPH_NODES,
		ProjectStorageValueLimits.ABSOLUTE_MAX_NODES
	)
	var graph_depth_budget: int = _read_bounded_integer(
		config,
		"max_graph_depth",
		DEFAULT_MAX_GRAPH_DEPTH,
		ProjectStorageValueLimits.ABSOLUTE_MAX_DEPTH
	)
	var string_byte_budget: int = _read_bounded_integer(
		config,
		"max_string_bytes",
		DEFAULT_MAX_STRING_BYTES,
		ProjectStorageValueLimits.ABSOLUTE_MAX_STRING_BYTES
	)
	var container_entry_budget: int = _read_bounded_integer(
		config,
		"max_container_entries",
		DEFAULT_MAX_CONTAINER_ENTRIES,
		ProjectStorageValueLimits.ABSOLUTE_MAX_CONTAINER_ENTRIES
	)
	if (
		payload_budget <= 0
		or list_item_budget <= 0
		or list_byte_budget <= 0
		or graph_node_budget <= 0
		or graph_depth_budget <= 0
		or string_byte_budget <= 0
		or container_entry_budget <= 0
	):
		return ERR_INVALID_PARAMETER

	var provider_options_value: Variant = config.get("provider_options", {})
	if not provider_options_value is Dictionary:
		return ERR_INVALID_PARAMETER
	var provider_options: Dictionary = provider_options_value
	var options_validation: Dictionary = (
		ProjectStorageValueLimits.validate_plain_value(
			provider_options,
			ProjectStorageValueLimits.make_limits(
				MAX_PROVIDER_OPTIONS_BYTES,
				4096,
				8,
				32 * 1024,
				1024
			)
		)
	)
	if not _validation_succeeded(options_validation):
		return _validation_error(options_validation)

	if candidate_provider.get_protocol_version() != PROTOCOL_VERSION:
		return ERR_INVALID_DATA
	var pre_initialize_capabilities: Dictionary = (
		candidate_provider.get_capabilities()
	)
	if not _provider_capabilities_are_valid(pre_initialize_capabilities):
		return ERR_INVALID_DATA

	var initialize_error: Error = candidate_provider.initialize(
		provider_options.duplicate(true)
	)
	var mapped_initialize_error: Error = _map_provider_error(
		int(initialize_error),
		ERR_UNAVAILABLE
	)
	if mapped_initialize_error != OK:
		candidate_provider.shutdown()
		return mapped_initialize_error

	if candidate_provider.get_protocol_version() != PROTOCOL_VERSION:
		candidate_provider.shutdown()
		return ERR_INVALID_DATA
	var post_initialize_capabilities: Dictionary = (
		candidate_provider.get_capabilities()
	)
	if (
		not _provider_capabilities_are_valid(post_initialize_capabilities)
		or post_initialize_capabilities != pre_initialize_capabilities
	):
		candidate_provider.shutdown()
		return ERR_INVALID_DATA

	_provider = candidate_provider
	_key_prefix = normalized_prefix
	_max_payload_bytes = payload_budget
	_max_list_items = list_item_budget
	_max_list_bytes = list_byte_budget
	_max_graph_nodes = graph_node_budget
	_max_graph_depth = graph_depth_budget
	_max_string_bytes = string_byte_budget
	_max_container_entries = container_entry_budget
	_supports_conditional_write = _get_bool(
		post_initialize_capabilities,
		"conditional_write"
	)
	_initialized = true
	return OK


## 关闭已接管 Provider 并恢复默认预算；解除引用后重复调用不会再次触达同一个 Provider。
## Provider.shutdown 返回后才清空引用；Provider 不得同步重入同一后端的关闭入口。
## [br]
## @api protected
func _shutdown() -> void:
	if _provider != null:
		_provider.shutdown()
	_provider = null
	_key_prefix = ""
	_max_payload_bytes = DEFAULT_MAX_PAYLOAD_BYTES
	_max_list_items = DEFAULT_MAX_LIST_ITEMS
	_max_list_bytes = DEFAULT_MAX_LIST_BYTES
	_max_graph_nodes = DEFAULT_MAX_GRAPH_NODES
	_max_graph_depth = DEFAULT_MAX_GRAPH_DEPTH
	_max_string_bytes = DEFAULT_MAX_STRING_BYTES
	_max_container_entries = DEFAULT_MAX_CONTAINER_ENTRIES
	_initialized = false
	_supports_conditional_write = false


## 基类保存钩子同样校验键和 payload，再由 Provider 负责原子提交，不能绕过普通保存入口的预算。
## [br]
## @api protected
## [br]
## @param file_name: 实例命名空间内的逻辑名称。
## [br]
## @param data: 待写入的业务数据。
## [br]
## @schema data: Dictionary，String 键的有界无循环纯值图。
## [br]
## @param metadata: 业务元数据和可选条件写信息。
## [br]
## @schema metadata: Dictionary，可含 write_options（expected_revision、create_if_absent、request_id），不得由调用方设置 revision；其余值遵守纯值预算。
## [br]
## @return 原子提交结果，未准备好为 ERR_UNAVAILABLE，非法键或选项与图预算失败均拒绝写入。
func _save_data(
	file_name: String,
	data: Dictionary,
	metadata: Dictionary
) -> Error:
	if not _is_ready():
		return ERR_UNAVAILABLE
	var storage_key: String = _resolve_storage_key(file_name)
	if storage_key.is_empty():
		return ERR_INVALID_PARAMETER

	var payload_validation: Dictionary = (
		ProjectStorageValueLimits.validate_payload(
			data,
			metadata,
			_make_payload_limits()
		)
	)
	if not _validation_succeeded(payload_validation):
		return _validation_error(payload_validation)
	return _save_validated_data(storage_key, data, metadata)


## 向 Provider 传递读取硬预算后复核完整响应；非法或矛盾响应只返回固定诊断和空数据，成功数据深复制。
## [br]
## @api protected
## [br]
## @param file_name: 实例命名空间内的逻辑名称。
## [br]
## @return 对调用者隔离的稳定读取结果。
## [br]
## @schema return: Dictionary，包含 ok:bool、data:Dictionary、metadata:Dictionary、error:String、error_code:int；失败数据为空，成功 metadata 在支持条件写时必须含合法 revision:String。
func _load_data(file_name: String) -> Dictionary:
	if not _is_ready():
		return _make_read_result(
			false,
			{},
			{},
			"Storage backend is unavailable.",
			ERR_UNAVAILABLE
		)
	var storage_key: String = _resolve_storage_key(file_name)
	if storage_key.is_empty():
		return _make_read_result(
			false,
			{},
			{},
			"Logical storage key is invalid.",
			ERR_INVALID_PARAMETER
		)

	var provider_result: Dictionary = _provider.read_record(
		storage_key,
		_make_read_limits()
	)
	if not _read_result_schema_is_valid(provider_result):
		return _make_read_result(
			false,
			{},
			{},
			"Storage provider returned an invalid result.",
			ERR_INVALID_DATA
		)
	var provider_ok: bool = _get_bool(provider_result, "ok")
	var provider_data: Dictionary = provider_result["data"]
	var provider_metadata: Dictionary = provider_result["metadata"]
	var provider_error_code: int = _get_int(
		provider_result,
		"error_code"
	)
	if not provider_ok:
		if (
			provider_error_code == int(OK)
			or not provider_data.is_empty()
			or not provider_metadata.is_empty()
		):
			return _make_read_result(
				false,
				{},
				{},
				"Storage provider returned an invalid result.",
				ERR_INVALID_DATA
			)
		return _make_read_result(
			false,
			{},
			{},
			"Storage provider read failed.",
			_map_provider_error(provider_error_code, ERR_CANT_OPEN)
		)
	if provider_error_code != int(OK):
		return _make_read_result(
			false,
			{},
			{},
			"Storage provider returned an invalid result.",
			ERR_INVALID_DATA
		)

	var payload_validation: Dictionary = (
		ProjectStorageValueLimits.validate_payload(
			provider_data,
			provider_metadata,
			_make_payload_limits()
		)
	)
	if not _validation_succeeded(payload_validation):
		return _make_read_result(
			false,
			{},
			{},
			"Storage provider read exceeded the accepted value contract.",
			_validation_error(payload_validation)
		)
	if not _record_metadata_is_valid(provider_metadata):
		return _make_read_result(
			false,
			{},
			{},
			"Storage provider returned invalid metadata.",
			ERR_INVALID_DATA
		)
	return _make_read_result(
		true,
		provider_data,
		provider_metadata,
		"",
		OK
	)


## 验证逻辑名称后删除对应 Provider 记录，只透出允许的稳定错误码。
## [br]
## @api protected
## [br]
## @param file_name: 实例命名空间内的逻辑名称。
## [br]
## @return 删除结果；未准备好或名称非法时不调用 Provider。
func _delete_data(file_name: String) -> Error:
	if not _is_ready():
		return ERR_UNAVAILABLE
	var storage_key: String = _resolve_storage_key(file_name)
	if storage_key.is_empty():
		return ERR_INVALID_PARAMETER
	var provider_error: Error = _provider.delete_record(storage_key)
	return _map_provider_error(int(provider_error), ERR_CANT_OPEN)


## 仅对已初始化 Provider 和合法逻辑键查询存在性；false 不区分未准备好、非法键和记录缺失。
## [br]
## @api protected
## [br]
## @param file_name: 实例命名空间内的逻辑名称。
## [br]
## @return Provider 报告存在时为 true，其余情况为 false。
func _has_data(file_name: String) -> bool:
	if not _is_ready():
		return false
	var storage_key: String = _resolve_storage_key(file_name)
	if storage_key.is_empty():
		return false
	return _provider.has_record(storage_key)


## 对完整列表检查聚合预算、命名空间、重复名称和修订，再按逻辑名称排序；任一失败均返回空列表。
## [br]
## @api protected
## [br]
## @return 有序元数据条目；空结果不能区分无记录与 Provider/预算失败。
## [br]
## @schema return: Array[Dictionary]，每项精确包含 file_name:String 和 metadata:Dictionary；metadata 为隔离副本，不含完整业务 payload。
func _list_data() -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	if not _is_ready():
		return result

	var provider_result: Dictionary = _provider.list_records(
		_key_prefix,
		_make_list_limits()
	)
	if not _list_result_schema_is_valid(provider_result):
		return result
	if not _get_bool(provider_result, "ok"):
		return result
	if _get_int(provider_result, "error_code") != int(OK):
		return result

	var records: Array = provider_result["records"]
	if records.size() > _max_list_items:
		return result
	var records_validation: Dictionary = (
		ProjectStorageValueLimits.validate_plain_value(
			records,
			_make_list_value_limits()
		)
	)
	if not _validation_succeeded(records_validation):
		return result

	var entries_by_name: Dictionary = {}
	var logical_names: PackedStringArray = PackedStringArray()
	for record_value: Variant in records:
		if not record_value is Dictionary:
			return []
		var record: Dictionary = record_value
		if not _has_exact_keys(record, _LIST_RECORD_KEYS):
			return []
		var storage_key_value: Variant = record.get("storage_key")
		var metadata_value: Variant = record.get("metadata")
		if not storage_key_value is String or not metadata_value is Dictionary:
			return []
		var storage_key: String = storage_key_value
		var metadata: Dictionary = metadata_value
		var logical_name: String = _logical_name_from_storage_key(storage_key)
		if (
			logical_name.is_empty()
			or entries_by_name.has(logical_name)
			or not _record_metadata_is_valid(metadata)
		):
			return []
		entries_by_name[logical_name] = {
			"file_name": logical_name,
			"metadata": metadata.duplicate(true),
		}
		var _append_result: bool = logical_names.append(logical_name)
	logical_names.sort()
	for logical_name: String in logical_names:
		var entry_value: Variant = entries_by_name.get(logical_name)
		if not entry_value is Dictionary:
			return []
		var entry: Dictionary = entry_value
		result.append(entry)
	return result


## 只报告初始化已提交的能力；关闭后撤销读写能力，本同步骨架始终不支持 sync 或 cancellation。
## [br]
## @api protected
## [br]
## @return 适配器当前可履行的闭合能力集合。
## [br]
## @schema return: Dictionary，protocol_version:String，以及 read、write、delete、list、sync、atomic_write、conditional_write、cancellation 八个 bool。
func _get_capabilities() -> Dictionary:
	return {
		"protocol_version": PROTOCOL_VERSION,
		"read": _initialized,
		"write": _initialized,
		"delete": _initialized,
		"list": _initialized,
		"sync": false,
		"atomic_write": _initialized,
		"conditional_write": (
			_initialized and _supports_conditional_write
		),
		"cancellation": false,
	}


# --- 框架内部方法 ---

## 在深复制或调用 Provider 之前校验逻辑键和完整 payload 图，拒绝循环值及超预算输入。
## [br]
## @api framework_internal
## [br]
## @param file_name: 不含实例前缀的便携逻辑名称。
## [br]
## @param data: 待提交的业务数据。
## [br]
## @schema data: Dictionary，String 键及无循环的有界纯值图，不允许对象、Callable、RID 或非有限浮点。
## [br]
## @param metadata: 业务元数据及本次写入条件。
## [br]
## @schema metadata: Dictionary，纯值图；可带 write_options 字典（expected_revision:String、create_if_absent:bool、request_id:String），不能自行设置 revision；write_options 不进入存储元数据。
## [br]
## @return 原子提交成功为 OK，准入、条件冲突或 Provider 失败为稳定 Error。
func save_data(
	file_name: String,
	data: Dictionary,
	metadata: Dictionary = {}
) -> Error:
	if file_name.is_empty():
		return ERR_INVALID_PARAMETER
	if not _is_ready():
		return ERR_UNAVAILABLE
	var storage_key: String = _resolve_storage_key(file_name)
	if storage_key.is_empty():
		return ERR_INVALID_PARAMETER
	var payload_validation: Dictionary = (
		ProjectStorageValueLimits.validate_payload(
			data,
			metadata,
			_make_payload_limits()
		)
	)
	if not _validation_succeeded(payload_validation):
		return _validation_error(payload_validation)
	return _save_validated_data(storage_key, data, metadata)


# --- 私有/辅助方法 ---

## 消费已通过 payload 预算的候选，剥离 write_options 并检查条件写能力；向 provider 传深复制容器，原子提交由 provider 负责。
## [br]
## @api private
func _save_validated_data(
	storage_key: String,
	data: Dictionary,
	metadata: Dictionary
) -> Error:
	if metadata.has("revision"):
		return ERR_INVALID_PARAMETER

	var write_options_value: Variant = metadata.get("write_options", {})
	if not write_options_value is Dictionary:
		return ERR_INVALID_PARAMETER
	var write_options: Dictionary = write_options_value
	if not _write_options_are_valid(write_options):
		return ERR_INVALID_PARAMETER
	if (
		(write_options.has("expected_revision")
			or _get_bool(write_options, "create_if_absent"))
		and not _supports_conditional_write
	):
		return ERR_UNAVAILABLE

	var stored_metadata: Dictionary = metadata.duplicate(true)
	var _erase_result: bool = stored_metadata.erase("write_options")
	var provider_error: Error = _provider.write_record_atomic(
		storage_key,
		data.duplicate(true),
		stored_metadata,
		write_options.duplicate(true)
	)
	return _map_provider_error(int(provider_error), ERR_CANT_CREATE)

## 只有初始化已提交且 provider 引用存在时允许操作。
## [br]
## @api private
func _is_ready() -> bool:
	return _initialized and _provider != null


## 验证便携逻辑名称后加实例前缀，非法名称返回空串。
## [br]
## @api private
func _resolve_storage_key(file_name: String) -> String:
	if not _is_valid_logical_name(file_name):
		return ""
	return _key_prefix + file_name


## 要求存储键属于当前前缀，并验证剥离后的逻辑名称；命名空间外或非法键返回空串。
## [br]
## @api private
func _logical_name_from_storage_key(storage_key: String) -> String:
	if not storage_key.begins_with(_key_prefix):
		return ""
	var logical_name: String = storage_key.substr(_key_prefix.length())
	return logical_name if _is_valid_logical_name(logical_name) else ""


## 允许空前缀，否则去掉一个尾斜线、验证逻辑路径后统一追加斜线，非法输入返回空串。
## [br]
## @api private
func _normalize_prefix(prefix: String) -> String:
	if prefix.is_empty():
		return ""
	var candidate: String = prefix.trim_suffix("/")
	if not _is_valid_logical_name(candidate):
		return ""
	return candidate + "/"


## 限制名称长度并拒绝控制码、空段、点段、设备名和跨平台非法字符，允许斜线分隔的逻辑层级。
## [br]
## @api private
func _is_valid_logical_name(file_name: String) -> bool:
	if (
		file_name.is_empty()
		or file_name.length() > MAX_LOGICAL_KEY_LENGTH
		or file_name.begins_with("/")
		or file_name.ends_with("/")
	):
		return false
	for index: int in range(file_name.length()):
		var code_point: int = file_name.unicode_at(index)
		if code_point < 32 or code_point == 127:
			return false
	for component: String in file_name.split("/", true):
		if (
			component.is_empty()
			or component == "."
			or component == ".."
			or component.ends_with(".")
			or component.ends_with(" ")
			or _is_reserved_component(component)
		):
			return false
		for index: int in range(component.length()):
			var character: String = component.substr(index, 1)
			if _FORBIDDEN_COMPONENT_CHARACTERS.contains(character):
				return false
	return true


## 按首个点之前的小写名称干判断 Windows 设备名，同时将上标 1、2、3 归一为普通数字。
## [br]
## @api private
func _is_reserved_component(component: String) -> bool:
	var first_dot: int = component.find(".")
	var component_stem: String = (
		component.substr(0, first_dot) if first_dot >= 0 else component
	)
	component_stem = (
		component_stem
		.to_lower()
		.replace("¹", "1")
		.replace("²", "2")
		.replace("³", "3")
	)
	return _RESERVED_COMPONENT_STEMS.has(component_stem)


## 限制写选项字段及不透明 token，create_if_absent 为真时不能同时指定 expected_revision。
## [br]
## @api private
func _write_options_are_valid(options: Dictionary) -> bool:
	if not _has_only_keys(options, _WRITE_OPTION_KEYS):
		return false
	if options.has("expected_revision"):
		var revision_value: Variant = options.get("expected_revision")
		if not revision_value is String:
			return false
		var revision: String = revision_value
		if not _is_valid_opaque_token(
			revision,
			MAX_REVISION_TOKEN_BYTES
		):
			return false
	if options.has("create_if_absent"):
		var create_value: Variant = options.get("create_if_absent")
		if not create_value is bool:
			return false
		var create_if_absent: bool = create_value
		if create_if_absent and options.has("expected_revision"):
			return false
	if options.has("request_id"):
		var request_id_value: Variant = options.get("request_id")
		if not request_id_value is String:
			return false
		var request_id: String = request_id_value
		if not _is_valid_opaque_token(
			request_id,
			MAX_REQUEST_ID_BYTES
		):
			return false
	return true


## 条件写能力开启时要求 metadata 带合法非空修订 token；未开启时不强制 revision。
## [br]
## @api private
func _record_metadata_is_valid(metadata: Dictionary) -> bool:
	if not _supports_conditional_write:
		return true
	var revision_value: Variant = metadata.get("revision")
	if not revision_value is String:
		return false
	var revision: String = revision_value
	return _is_valid_opaque_token(
		revision,
		MAX_REVISION_TOKEN_BYTES
	)


## 检查非空 token 的 UTF-8 字节上限和控制码，不解读 token 内容或将其转成数字。
## [br]
## @api private
func _is_valid_opaque_token(value: String, maximum_bytes: int) -> bool:
	if value.is_empty() or value.length() > maximum_bytes:
		return false
	var byte_count: int = value.to_utf8_buffer().size()
	if byte_count > maximum_bytes:
		return false
	for index: int in range(value.length()):
		var code_point: int = value.unicode_at(index)
		if code_point < 32 or code_point == 127:
			return false
	return true


## 要求能力精确闭合且全部布尔，读写删列及原子写必须支持，取消和同步必须为 false。
## [br]
## @api private
func _provider_capabilities_are_valid(capabilities: Dictionary) -> bool:
	if not _has_exact_keys(capabilities, _CAPABILITY_KEYS):
		return false
	for capability_id: String in _CAPABILITY_KEYS:
		if not capabilities.get(capability_id) is bool:
			return false
	for required_id: String in [
		"read",
		"write",
		"delete",
		"list",
		"atomic_write",
	]:
		if not _get_bool(capabilities, required_id):
			return false
	return (
		not _get_bool(capabilities, "cancellation")
		and not _get_bool(capabilities, "sync")
	)


## 检查 provider 读取结果精确字段及基础类型，数据图预算和成功状态一致性由调用者继续验证。
## [br]
## @api private
func _read_result_schema_is_valid(result: Dictionary) -> bool:
	return (
		_has_exact_keys(result, _READ_RESULT_KEYS)
		and result.get("ok") is bool
		and result.get("data") is Dictionary
		and result.get("metadata") is Dictionary
		and typeof(result.get("error_code")) == TYPE_INT
	)


## 检查 provider 列表信封的精确字段和基础类型，记录及总响应预算另行验证。
## [br]
## @api private
func _list_result_schema_is_valid(result: Dictionary) -> bool:

		and result.get("ok") is bool
		and result.get("records") is Array
		and typeof(result.get("error_code")) == TYPE_INT
	)


## 读取配置整数或默认值，非 int、非正数或超过绝对上限返回 -1。
## [br]
## @api private
func _read_bounded_integer(
	values: Dictionary,
	field_name: String,
	default_value: int,
	absolute_maximum: int
) -> int:

	if not value is int:

	if integer_value <= 0 or integer_value > absolute_maximum:
		return -1
	return integer_value

## 只透传允许集合中的 int 错误码，其他类型或未知数值使用操作指定的稳定回退码。
## [br]
## @api private
func _map_provider_error(
	error_value: Variant,
	fallback: Error
) -> Error:
	if not error_value is int:
		return fallback
	var numeric_error: int = error_value
	if not _ALLOWED_PROVIDER_ERRORS.has(numeric_error):
		return fallback
	return numeric_error as Error


## 严格要求验证结果 ok 为真且 error_code 为整数 OK，拒绝矛盾或不完整结果。
## [br]
## @api private
func _validation_succeeded(validation: Dictionary) -> bool:
	return (
		validation.get("ok") is bool
		and _get_bool(validation, "ok")
		and typeof(validation.get("error_code")) == TYPE_INT
		and _get_int(validation, "error_code") == int(OK)
	)


## 仅保留验证器的资源超限错误，其余失败统一映射为 ERR_INVALID_DATA。
## [br]
## @api private
func _validation_error(validation: Dictionary) -> Error:
	if typeof(validation.get("error_code")) != TYPE_INT:
		return ERR_INVALID_DATA
	var error_code: int = _get_int(validation, "error_code")
	if error_code == int(ERR_OUT_OF_MEMORY):
		return ERR_OUT_OF_MEMORY
	return ERR_INVALID_DATA


## 拒绝未知或非文本键后，再要求所有必需键存在。
## [br]
## @api private
func _has_exact_or_optional_keys(
	values: Dictionary,
	allowed_keys: PackedStringArray,
	required_keys: PackedStringArray
) -> bool:
	if not _has_only_keys(values, allowed_keys):
		return false
	for required_key: String in required_keys:
		if not values.has(required_key):
			return false
	return true


## 用字段数量及允许键检查构成闭合字典校验。
## [br]
## @api private
func _has_exact_keys(
	values: Dictionary,
	expected_keys: PackedStringArray
) -> bool:
	return (
		values.size() == expected_keys.size()
		and _has_only_keys(values, expected_keys)
	)


## 要求所有键均为 String 且属于白名单，不要求可选字段出现。
## [br]
## @api private
func _has_only_keys(
	values: Dictionary,
	allowed_keys: PackedStringArray
) -> bool:
	for key_value: Variant in values.keys():
		if not key_value is String:
			return false
		var key: String = key_value
		if not allowed_keys.has(key):
			return false
	return true


## 读取精确 bool 字段，缺失或类型不符返回 false。
## [br]
## @api private
func _get_bool(values: Dictionary, field_name: String) -> bool:
	var value: Variant = values.get(field_name)
	if not value is bool:
		return false
	var boolean_value: bool = value
	return boolean_value


## 读取精确 int 字段，缺失或类型不符返回零。
## [br]
## @api private
func _get_int(values: Dictionary, field_name: String) -> int:
	var value: Variant = values.get(field_name)
	if not value is int:
		return 0
	var integer_value: int = value
	return integer_value
