## 存储适配边界共用的有界纯值验证器，只接受 null、bool、int、有限 float、String、Array 和 String 键 Dictionary。
## 接受范围刻意小于 Variant；验证过程中不把被检查值转成字符串或写入日志。
## [br]
## @api framework_internal
class_name ProjectStorageValueLimits
extends RefCounted


# --- 常量 ---

## 单次纯值图保守字节计费的不可关闭上限，调用方只能请求更小预算。
## [br]
## @api framework_internal
const ABSOLUTE_MAX_BYTES: int = 64 * 1024 * 1024

## 单次遍历允许计费的最大结构节点数，避免大量小值绕过字节限制。
## [br]
## @api framework_internal
const ABSOLUTE_MAX_NODES: int = 262_144

## 从根深度零开始计算的最大嵌套深度，防止无界递归。
## [br]
## @api framework_internal
const ABSOLUTE_MAX_DEPTH: int = 32

## 单条 String 的 UTF-8 字节绝对上限，独立于整个值图的预算。
## [br]
## @api framework_internal
const ABSOLUTE_MAX_STRING_BYTES: int = 4 * 1024 * 1024

## 单个 Array 或 Dictionary 的最大元素数，包含列表响应的外层数组。
## [br]
## @api framework_internal
const ABSOLUTE_MAX_CONTAINER_ENTRIES: int = 65_536

## 纯数据预算字典的精确字段集合，缺失或额外字段都拒绝准入。
## [br]
## @api private
const _LIMIT_KEYS: PackedStringArray = [
	"max_bytes",
	"max_nodes",
	"max_depth",
	"max_string_bytes",
	"max_container_entries",
]


# --- 框架内部方法 ---

## 构造闭合预算字典；这里只组装字段，正整数和绝对上限由验证入口统一检查。
## [br]
## @api framework_internal
## [br]
## @param max_bytes: 整个值图的保守字节预算。
## [br]
## @param max_nodes: 整个值图的节点数预算。
## [br]
## @param max_depth: 从根深度零计算的嵌套预算。
## [br]
## @param max_string_bytes: 每条文本的 UTF-8 字节预算。
## [br]
## @param max_container_entries: 每个普通容器的元素数预算。
## [br]
## @return 原样记录五项输入的预算字典。
## [br]
## @schema return: Dictionary，精确包含 max_bytes、max_nodes、max_depth、max_string_bytes、max_container_entries 五个 int 字段。
static func make_limits(
	max_bytes: int,
	max_nodes: int,
	max_depth: int,
	max_string_bytes: int,
	max_container_entries: int
) -> Dictionary:
	return {
		"max_bytes": max_bytes,
		"max_nodes": max_nodes,
		"max_depth": max_depth,
		"max_string_bytes": max_string_bytes,
		"max_container_entries": max_container_entries,
	}


## 不序列化或打印输入，按有界纯值图规则遍历；拒绝对象、非有限浮点和循环容器，保留失败时已计费的统计。
## [br]
## @api framework_internal
## [br]
## @param value: 待验证的任意输入，非法类型返回失败而非隐式转换。
## [br]
## @schema value: Variant，仅允许 null、bool、int、有限 float、String、Array 及 String 键 Dictionary 递归组成的无循环值图。
## [br]
## @param limits: 本次验证的精确预算字段。
## [br]
## @schema limits: Dictionary，max_bytes、max_nodes、max_depth、max_string_bytes、max_container_entries 均为不超过各 ABSOLUTE 上限的正 int。
## [br]
## @return 成功或失败及已消耗的保守预算，不能把 size_bytes 当实际序列化长度。
## [br]
## @schema return: Dictionary，精确包含 ok:bool、error_code:int、size_bytes:int、node_count:int；预算非法为 ERR_INVALID_PARAMETER，超限为 ERR_OUT_OF_MEMORY，值类型或循环非法为 ERR_INVALID_DATA。
static func validate_plain_value(value: Variant, limits: Dictionary) -> Dictionary:
	if not _limits_are_valid(limits):
		return _make_result(false, ERR_INVALID_PARAMETER, 0, 0)

	var state: Dictionary = {
		"bytes": 0,
		"nodes": 0,
	}
	var ancestors: Array[Variant] = []
	var validation_error: Error = _inspect_value(
		value,
		0,
		limits,
		state,
		ancestors
	)
	return _make_result(
		validation_error == OK,
		validation_error,
		_get_int(state, "bytes"),
		_get_int(state, "nodes")
	)


## 将 data 和 metadata 放入同一信封共同计费，避免各自合规却合计超限。
## [br]
## @api framework_internal
## [br]
## @param data: 待验证的业务数据。
## [br]
## @schema data: Dictionary，须符合 validate_plain_value 的 String 键纯值图规则。
## [br]
## @param metadata: 与业务数据一起提交的元数据。
## [br]
## @schema metadata: Dictionary，遵守与 data 相同的纯值图规则。
## [br]
## @param limits: 整个信封共享的预算。
## [br]
## @schema limits: Dictionary，精确包含 max_bytes、max_nodes、max_depth、max_string_bytes、max_container_entries 五个正 int。
## [br]
## @return 对 {data,metadata} 信封的验证结果。
## [br]
## @schema return: Dictionary，包含 ok:bool、error_code:int、size_bytes:int、node_count:int，与 validate_plain_value 一致。
static func validate_payload(
	data: Dictionary,
	metadata: Dictionary,
	limits: Dictionary
) -> Dictionary:
	var payload: Dictionary = {
		"data": data,
		"metadata": metadata,
	}
	return validate_plain_value(payload, limits)


# --- 私有/辅助方法 ---

## 要求预算字段精确闭合且均为正整数，每项不得超过不可关闭的绝对上限。
## [br]
## @api private
static func _limits_are_valid(limits: Dictionary) -> bool:
	if not _has_exact_keys(limits, _LIMIT_KEYS):
		return false
	return (
		_is_bounded_integer(
			limits.get("max_bytes"),
			ABSOLUTE_MAX_BYTES
		)
		and _is_bounded_integer(
			limits.get("max_nodes"),
			ABSOLUTE_MAX_NODES
		)
		and _is_bounded_integer(
			limits.get("max_depth"),
			ABSOLUTE_MAX_DEPTH
		)
		and _is_bounded_integer(
			limits.get("max_string_bytes"),
			ABSOLUTE_MAX_STRING_BYTES
		)
		and _is_bounded_integer(
			limits.get("max_container_entries"),
			ABSOLUTE_MAX_CONTAINER_ENTRIES
		)
	)


## 只接受精确 int，值须大于零且不超过传入上限，拒绝浮点形式的整数。
## [br]
## @api private
static func _is_bounded_integer(value: Variant, absolute_maximum: int) -> bool:
	if not value is int:
		return false
	var integer_value: int = value
	return integer_value > 0 and integer_value <= absolute_maximum


## 先检查深度并累计节点，再按类型保守计费；只接受有限标量、文本及普通容器，失败保留已消耗统计。
## [br]
## @api private
static func _inspect_value(
	value: Variant,
	depth: int,
	limits: Dictionary,
	state: Dictionary,
	ancestors: Array[Variant]
) -> Error:
	if depth > _get_int(limits, "max_depth"):
		return ERR_OUT_OF_MEMORY
	var node_error: Error = _add_node(
		state,
		_get_int(limits, "max_nodes")
	)
	if node_error != OK:
		return node_error

	if value == null:
		return _add_bytes(state, 4, _get_int(limits, "max_bytes"))
	if value is bool:
		return _add_bytes(state, 5, _get_int(limits, "max_bytes"))
	if value is int:
		return _add_bytes(state, 32, _get_int(limits, "max_bytes"))
	if value is float:
		var float_value: float = value
		if not is_finite(float_value):
			return ERR_INVALID_DATA
		return _add_bytes(state, 64, _get_int(limits, "max_bytes"))
	if value is String:
		var text: String = value
		return _add_string_bytes(text, limits, state)
	if value is Array:
		var array_value: Array = value
		return _inspect_array(array_value, depth, limits, state, ancestors)
	if value is Dictionary:
		var dictionary_value: Dictionary = value
		return _inspect_dictionary(
			dictionary_value,
			depth,
			limits,
			state,
			ancestors
		)
	return ERR_INVALID_DATA


## 限制数组条目数并以祖先身份拒绝循环，累计括号、分隔符和子值预算；已压栈的退出路径都会弹栈。
## [br]
## @api private
static func _inspect_array(
	values: Array,
	depth: int,
	limits: Dictionary,
	state: Dictionary,
	ancestors: Array[Variant]
) -> Error:
	if values.size() > _get_int(limits, "max_container_entries"):
		return ERR_OUT_OF_MEMORY
	if _contains_same_container(ancestors, values):
		return ERR_INVALID_DATA
	var open_error: Error = _add_bytes(
		state,
		2,
		_get_int(limits, "max_bytes")
	)
	if open_error != OK:
		return open_error
	ancestors.append(values)
	for item: Variant in values:
		var separator_error: Error = _add_bytes(
			state,
			1,
			_get_int(limits, "max_bytes")
		)
		if separator_error != OK:
			var _discarded_separator_ancestor: Variant = ancestors.pop_back()
			return separator_error
		var item_error: Error = _inspect_value(
			item,
			depth + 1,
			limits,
			state,
			ancestors
		)
		if item_error != OK:
			var _discarded_item_ancestor: Variant = ancestors.pop_back()
			return item_error
	var _discarded_ancestor: Variant = ancestors.pop_back()
	return OK


## 限制条目数、拒绝循环及非 String 键，键和值分别计费；进入后所有失败路径都恢复祖先栈。
## [br]
## @api private
static func _inspect_dictionary(
	values: Dictionary,
	depth: int,
	limits: Dictionary,
	state: Dictionary,
	ancestors: Array[Variant]
) -> Error:
	if values.size() > _get_int(limits, "max_container_entries"):
		return ERR_OUT_OF_MEMORY
	if _contains_same_container(ancestors, values):
		return ERR_INVALID_DATA
	var open_error: Error = _add_bytes(
		state,
		2,
		_get_int(limits, "max_bytes")
	)
	if open_error != OK:
		return open_error
	ancestors.append(values)
	for key_value: Variant in values.keys():
		if not key_value is String:
			var _discarded_key_ancestor: Variant = ancestors.pop_back()
			return ERR_INVALID_DATA
		var key: String = key_value
		var separator_error: Error = _add_bytes(
			state,
			2,
			_get_int(limits, "max_bytes")
		)
		if separator_error != OK:
			var _discarded_separator_ancestor: Variant = ancestors.pop_back()
			return separator_error
		var key_error: Error = _add_string_bytes(key, limits, state)
		if key_error != OK:
			var _discarded_string_ancestor: Variant = ancestors.pop_back()
			return key_error
		var value_error: Error = _inspect_value(
			values.get(key),
			depth + 1,
			limits,
			state,
			ancestors
		)
		if value_error != OK:
			var _discarded_value_ancestor: Variant = ancestors.pop_back()
			return value_error
	var _discarded_ancestor: Variant = ancestors.pop_back()
	return OK


## 按引用身份检查当前祖先链，只拒绝循环，其他分支复用容器不受影响。
## [br]
## @api private
static func _contains_same_container(
	ancestors: Array[Variant],
	candidate: Variant
) -> bool:
	for ancestor: Variant in ancestors:
		if is_same(ancestor, candidate):
			return true
	return false


## 先检查实际 UTF-8 字符串上限，再以六倍字节加引号保守预留转义后的总大小。
## [br]
## @api private
static func _add_string_bytes(
	value: String,
	limits: Dictionary,
	state: Dictionary
) -> Error:
	var maximum_string_bytes: int = _get_int(
		limits,
		"max_string_bytes"
	)
	if value.length() > maximum_string_bytes:
		return ERR_OUT_OF_MEMORY
	var utf8_size: int = value.to_utf8_buffer().size()
	if utf8_size > maximum_string_bytes:
		return ERR_OUT_OF_MEMORY
	if utf8_size * 6 > _get_int(limits, "max_bytes") - 2:
		return ERR_OUT_OF_MEMORY
	return _add_bytes(
		state,
		utf8_size * 6 + 2,
		_get_int(limits, "max_bytes")
	)


## 预留一个结构节点，超限返回 ERR_OUT_OF_MEMORY 且不增加计数。
## [br]
## @api private
static func _add_node(state: Dictionary, maximum: int) -> Error:
	var next_count: int = _get_int(state, "nodes") + 1
	if next_count > maximum:
		return ERR_OUT_OF_MEMORY
	state["nodes"] = next_count
	return OK


## 拒绝负追加量或超限，只有成功时累加共享字节统计。
## [br]
## @api private
static func _add_bytes(
	state: Dictionary,
	additional_bytes: int,
	maximum: int
) -> Error:
	if (
		additional_bytes < 0
		or _get_int(state, "bytes") > maximum - additional_bytes
	):
		return ERR_OUT_OF_MEMORY
	state["bytes"] = _get_int(state, "bytes") + additional_bytes
	return OK


## 要求字典字段数量匹配且所有键为允许的精确 String。
## [br]
## @api private
static func _has_exact_keys(
	values: Dictionary,
	expected_keys: PackedStringArray
) -> bool:
	if values.size() != expected_keys.size():
		return false
	for key_value: Variant in values.keys():
		if not key_value is String:
			return false
		var key: String = key_value
		if not expected_keys.has(key):
			return false
	return true


## 读取精确 int 字段，缺失或其他类型返回零，供已验证预算和计数使用。
## [br]
## @api private
static func _get_int(values: Dictionary, field_name: String) -> int:
	var value: Variant = values.get(field_name)
	if not value is int:
		return 0
	var integer_value: int = value
	return integer_value


## 构造闭合验证结果，保留成功或失败时已累计的保守字节数和节点数。
## [br]
## @api private
static func _make_result(
	ok: bool,
	error_code: Error,
	size_bytes: int,
	node_count: int
) -> Dictionary:
	return {
		"ok": ok,
		"error_code": int(error_code),
		"size_bytes": size_bytes,
		"node_count": node_count,
	}
